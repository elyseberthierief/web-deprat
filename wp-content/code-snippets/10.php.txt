function deprat_adv_est_demande( $commande ) {
    return $commande instanceof WC_Order && 'oui' === $commande->get_meta( '_deprat_demande', true );
}
function deprat_adv_ordre_courant() {
    if ( ! function_exists( 'wc_get_order' ) || ! is_admin() ) { return false; }
    $id = 0;
    if ( isset( $_GET['page'], $_GET['action'], $_GET['id'] ) &&
         'wc-orders' === sanitize_key( wp_unslash( $_GET['page'] ) ) &&
         'edit' === sanitize_key( wp_unslash( $_GET['action'] ) ) ) {
        $id = absint( $_GET['id'] );
    } elseif ( isset( $_GET['post'], $_GET['action'] ) &&
         'edit' === sanitize_key( wp_unslash( $_GET['action'] ) ) ) {
        $id = absint( $_GET['post'] );
    }
    if ( ! $id ) { return false; }
    $commande = wc_get_order( $id );
    return deprat_adv_est_demande( $commande ) ? $commande : false;
}
function deprat_adv_date_heure( $utc ) {
    if ( ! $utc ) { return '—'; }
    if ( function_exists( 'deprat_date_utc_v7' ) ) { return deprat_date_utc_v7( $utc ); }
    $timestamp = strtotime( $utc . ' UTC' );
    return $timestamp ? wp_date( 'd/m/Y à H:i', $timestamp ) : '—';
}
function deprat_adv_nom_etat( $etat ) {
    $noms = array(
        'dpr-recu' => 'Nouvelle demande',
        'dpr-encours' => 'En cours de traitement',
        'dpr-envoye' => 'Envoyer le devis',
        'dpr-accepte' => 'Devis accepté',
        'dpr-facture' => 'Envoyer la facture',
        'dpr-termine' => 'Commande terminée',
    );
    return $noms[ $etat ] ?? $etat;
}

// Encadré au-dessus du formulaire de documents V7.3.
add_action( 'woocommerce_admin_order_data_after_order_details', function( $commande ) {
    if ( ! deprat_adv_est_demande( $commande ) ) { return; }
    if ( ! current_user_can( 'manage_woocommerce' ) && ! current_user_can( 'edit_shop_orders' ) ) { return; }
    $accord = (string) $commande->get_meta( '_deprat_devis_accepte_le', true );
    $facture = (string) $commande->get_meta( '_deprat_facture_envoyee_le', true );
    $revision = 'oui' === $commande->get_meta( '_deprat_revision_en_attente', true );
    $commentaire = (string) $commande->get_meta( '_deprat_commentaire', true );
    echo '<section class="deprat-adv-bandeau" aria-label="Tableau de bord ADV">';
    echo '<div class="deprat-adv-bandeau-titre"><div><span class="deprat-adv-surtitre">DEPRAT · SERVICE ADV</span><h2>Suivi de la demande n° ' . esc_html( $commande->get_order_number() ) . '</h2></div>';
    echo '<span class="deprat-adv-statut">' . esc_html( deprat_adv_nom_etat( $commande->get_status() ) ) . '</span></div>';
    echo '<div class="deprat-adv-indicateurs">';
    echo '<div><small>Société</small><strong>' . esc_html( $commande->get_billing_company() ?: 'Non renseignée' ) . '</strong></div>';
    echo '<div><small>Accord du client</small><strong>' . ( $accord ? esc_html( deprat_adv_date_heure( $accord ) ) : 'En attente' ) . '</strong></div>';
    echo '<div><small>Facture disponible</small><strong>' . ( $facture ? esc_html( deprat_adv_date_heure( $facture ) ) : 'Non publiée' ) . '</strong></div>';
    echo '</div>';
    if ( $commentaire ) {
        echo '<div class="deprat-adv-commentaire"><strong>Commentaire de la demande</strong><p>' . nl2br( esc_html( $commentaire ) ) . '</p></div>';
    }
    if ( $revision ) {
        echo '<div class="deprat-adv-warning deprat-adv-revision"><strong>Le client a demandé une modification du devis.</strong><p>' . nl2br( esc_html( (string) $commande->get_meta( '_deprat_dernier_ajustement_texte', true ) ) ) . '</p><p>Déposez un nouveau devis avant de choisir « Envoyer le devis ».</p></div>';
    }
    if ( ! $accord ) {
        echo '<div class="deprat-adv-warning" id="deprat-adv-non-valide"><strong>Facturation impossible pour le moment.</strong> Le client n’a pas accepté son devis dans l’espace professionnel. N’utilisez pas « Envoyer la facture ».</div>';
    }
    echo '</section>';
}, 5 );

// Tableau des références sous les articles natifs WooCommerce (qui restent enregistrés sans modification).
add_action( 'woocommerce_admin_order_items_after_line_items', function( $order_id ) {
    $commande = wc_get_order( absint( $order_id ) );
    if ( ! deprat_adv_est_demande( $commande ) ) { return; }
    echo '<tr class="deprat-adv-lignes-produits"><td colspan="8">';
    echo '<div class="deprat-adv-articles-entete"><h3>Références demandées</h3><span>' . esc_html( count( $commande->get_items( 'line_item' ) ) ) . ' ligne(s)</span></div>';
    echo '<div class="deprat-adv-articles-scroll"><table class="deprat-adv-table-produits"><thead><tr><th>Référence</th><th>Désignation</th><th>Quantité</th></tr></thead><tbody>';
    foreach ( $commande->get_items( 'line_item' ) as $ligne ) {
        $ref = (string) $ligne->get_meta( '_deprat_reference', true );
        if ( ! $ref && $ligne->get_product() ) { $ref = (string) $ligne->get_product()->get_sku(); }
        echo '<tr><td><code>' . esc_html( $ref ?: 'Sans référence' ) . '</code></td><td>' . esc_html( $ligne->get_name() ) . '</td><td class="deprat-adv-quantite">' . esc_html( $ligne->get_quantity() ) . '</td></tr>';
    }
    echo '</tbody></table></div></td></tr>';
}, 10, 1 );

// Réafficher les notes en timeline lisible, consultable sans supprimer depuis cette zone.
add_action( 'woocommerce_admin_order_data_after_order_details', function( $commande ) {
    if ( ! deprat_adv_est_demande( $commande ) || ! function_exists( 'wc_get_order_notes' ) ) { return; }
    $notes = wc_get_order_notes( array( 'order_id' => $commande->get_id(), 'limit' => 60, 'orderby' => 'date_created', 'order' => 'DESC' ) );
    echo '<section class="deprat-adv-historique" aria-label="Historique de la demande"><div class="deprat-adv-historique-titre"><h3>Historique de la demande</h3><span>' . esc_html( count( $notes ) ) . ' derniers événements</span></div>';
    echo '<p class="deprat-adv-note-aide">Événements WooCommerce et actions du service ADV, du plus récent au plus ancien. Les notes ne sont pas effaçables depuis cette interface.</p>';
    if ( ! $notes ) { echo '<p>Pas encore d’événement.</p>'; }
    else {
        echo '<ol class="deprat-adv-timeline">';
        foreach ( $notes as $note ) {
            $date = $note->date_created instanceof WC_DateTime ? wp_date( 'd/m/Y à H:i', $note->date_created->getTimestamp() ) : '';
            echo '<li><time>' . esc_html( $date ) . '</time><div>' . wp_kses_post( wpautop( $note->content ) ) . '</div></li>';
        }
        echo '</ol>';
    }
    echo '</section>';
}, 45 );

// Bloquer le bouton AJAX « Supprimer la note » pour les demandes DEPRAT.
// C'est un garde-fou ciblé sur l'interface WooCommerce ; sauvegardes/permissions restent nécessaires.
$deprat_bloquer_suppression_note = function () {
    $note_id = isset( $_POST['note_id'] ) ? absint( $_POST['note_id'] ) : 0;
    $commentaire = $note_id ? get_comment( $note_id ) : false;
    if ( ! $commentaire || 'order_note' !== $commentaire->comment_type ) { return; }
    $commande = wc_get_order( (int) $commentaire->comment_post_ID );
    if ( deprat_adv_est_demande( $commande ) ) {
        wp_die( 'Suppression désactivée pour l’historique des demandes DEPRAT.', 'Historique protégé', array( 'response' => 403 ) );
    }
};
add_action( 'wp_ajax_woocommerce_delete_order_note', $deprat_bloquer_suppression_note, 0 );
add_action( 'wc_ajax_delete_order_note', $deprat_bloquer_suppression_note, 0 );
